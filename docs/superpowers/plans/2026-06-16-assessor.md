# Assessor Implementation Plan (Plan 4)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Anchored generative probes (rubric-grounded self-explanation, calibration, artifact review, source quiz) that detect the illusion of competence, write an anchored `Assessment` with a delta, **update the `KnowledgeEstimate`** (moving the Tutor's assistance/challenge levers), and propose a thin curated SM-2 retention layer.

**Architecture:** Pure helpers (`sm2_schedule`, `level_from_score`, `compute_delta`, `apply_rules_checks`, `penalize_score`, JSON parsers) + lazy `ensure_rubric`/`ensure_anchor` + a single-LLM-call `assess_concept` orchestrator that persists, traces via `AgentEvent`, and updates the estimate. Mirrors `engine/curriculum.py` / `engine/research.py`.

**Tech Stack:** Python 3.12 · uv · FastAPI · SQLModel/SQLite · Alembic · SvelteKit 5 / Tailwind 4 / vitest.

**Spec:** `docs/superpowers/specs/2026-06-16-assessor.md`. Read §2 (decisions) and §3 (architecture) before starting.

**Conventions (match existing code):** JSON columns use `sa_column=Column(JSON)`. Engine orchestrators take injected `llm`, persist, emit topic-scoped `AgentEvent`s (`conversation_id=None`, `topic_id=<concept's topic>`), return a result dataclass, then refresh. Tests use the `session` fixture (`tests/conftest.py`) and `ScriptedLLMClient`. Keep all imports at module top. Run backend tests with `uv run pytest -q`, lint with `uv run ruff check src tests`.

**Topic id from a concept:** a `Concept` → `Track` → `Topic`. Helper: `session.get(Track, concept.track_id).topic_id`.

# src/adaptive_academy/engine/retention.py
_QUALITY = {"again": 2, "hard": 3, "good": 4, "easy": 5}


def sm2_schedule(ease: float, interval: int, reps: int, result: str) -> tuple[float, int, int]:
    """Pure SM-2 step. Returns the new (ease, interval_days, reps).

    A lapse (again, q<3) resets interval to 1 and reps to 0; the ease factor is
    adjusted by the standard SM-2 formula for every grade and floored at 1.3.
    """
    q = _QUALITY.get(result, 4)
    new_ease = max(1.3, ease + (0.1 - (5 - q) * (0.08 + (5 - q) * 0.02)))
    if q < 3:
        return new_ease, 1, 0
    new_reps = reps + 1
    if new_reps == 1:
        new_interval = 1
    elif new_reps == 2:
        new_interval = 6
    else:
        new_interval = max(1, round(interval * new_ease))
    return new_ease, new_interval, new_reps
```

# src/adaptive_academy/engine/assess.py
_OMISSION_PENALTY = 0.15


def level_from_score(score: float) -> tuple[str, float]:
    """Map an anchored score (0..1) to a KnowledgeEstimate level + confidence.

    Thresholds are empirical (tune here): >=0.75 high, >=0.45 medium, else low.
    """
    s = max(0.0, min(1.0, score))
    level = "high" if s >= 0.75 else "medium" if s >= 0.45 else "low"
    return level, round(s, 2)


def compute_delta(score: float, baseline_score: float | None) -> float | None:
    if baseline_score is None:
        return None
    return round(score - baseline_score, 4)


def apply_rules_checks(response: str, must_include: list[str], weak_signs: list[str]) -> list[str]:
    """Deterministic anchor checks (§2.3): the LLM grader cannot paper over these."""
    text = response.lower()
    gaps: list[str] = []
    for term in must_include:
        if term and term.lower() not in text:
            gaps.append(f"omitted required element: {term}")
    for sign in weak_signs:
        if sign and sign.lower() in text:
            gaps.append(f"contains likely misconception: {sign}")
    return gaps


def penalize_score(llm_score: float, rules_gaps: list[str]) -> float:
    """Penalize the LLM score for each omitted required element (anti-sycophancy)."""
    omissions = sum(1 for g in rules_gaps if g.startswith("omitted required element"))
    return round(max(0.0, min(1.0, llm_score) - _OMISSION_PENALTY * omissions), 4)
```

import json
from dataclasses import dataclass, field


@dataclass
class RubricDraft:
    criteria: list[str] = field(default_factory=list)
    must_include: list[str] = field(default_factory=list)
    strong_signs: list[str] = field(default_factory=list)
    weak_signs: list[str] = field(default_factory=list)
    source_refs: list[str] = field(default_factory=list)


@dataclass
class GradeDraft:
    score: float = 0.0
    verdict: str = ""
    gaps: list[str] = field(default_factory=list)
    confidence_actual: float | None = None


def _strip_fences(text: str) -> str:
    cleaned = text.strip()
    if cleaned.startswith("```"):
        cleaned = cleaned.split("```", 2)[1]
        if cleaned.startswith("json"):
            cleaned = cleaned[len("json"):]
    return cleaned.strip()


def _as_float(v, default: float | None) -> float | None:
    try:
        return float(v)
    except (TypeError, ValueError):
        return default


def parse_rubric_json(text: str) -> RubricDraft:
    d = json.loads(_strip_fences(text))
    return RubricDraft(
        criteria=list(d.get("criteria", []) or []),
        must_include=list(d.get("must_include", []) or []),
        strong_signs=list(d.get("strong_signs", []) or []),
        weak_signs=list(d.get("weak_signs", []) or []),
        source_refs=list(d.get("source_refs", []) or []),
    )


def parse_grade_json(text: str) -> GradeDraft:
    d = json.loads(_strip_fences(text))
    return GradeDraft(
        score=_as_float(d.get("score"), 0.0) or 0.0,
        verdict=str(d.get("verdict", "")),
        gaps=list(d.get("gaps", []) or []),
        confidence_actual=_as_float(d.get("confidence_actual"), None),
    )
```

from sqlmodel import Session, select

from adaptive_academy.llm.types import LLMClient
from adaptive_academy.models import (
    Concept, Rubric, SourceDoc, TopicFoundation, Track,
)

_RUBRIC_SYSTEM = (
    "You build a grading rubric for one concept an advanced learner must explain. "
    "Given the concept, the topic foundation, and sources, produce a rubric to grade a "
    "learner's *explanation* against — NOT a model answer. Respond with ONLY a JSON object: "
    '{"criteria": [str, prose points a correct explanation must address], '
    '"must_include": [str, short LITERAL terms/expressions a correct explanation must '
    'contain, e.g. "softmax", "QK^T", "scale by sqrt(d)"], "strong_signs": [str], '
    '"weak_signs": [str, common misconceptions/red flags], "source_refs": [str]}. '
    "Keep must_include to 3-6 precise, checkable terms. No prose outside the JSON."
)


def _topic_id_for(session: Session, concept: Concept) -> int:
    return session.get(Track, concept.track_id).topic_id


def build_rubric_prompt(concept_name: str, foundation: TopicFoundation | None,
                        sources: list[SourceDoc]) -> tuple[str, str]:
    lines = [f"CONCEPT: {concept_name}"]
    if foundation:
        lines.append(f"FOUNDATION OVERVIEW: {foundation.overview}")
        lines.append(f"KEY CONCEPTS: {', '.join(foundation.key_concepts)}")
    if sources:
        lines.append("SOURCES:")
        for s in sources:
            lines.append(f"- [{s.tier}] {s.title} — {s.summary}".rstrip(" —"))
    return _RUBRIC_SYSTEM, "\n".join(lines)


def ensure_rubric(session: Session, concept: Concept, *, llm: LLMClient) -> Rubric:
    """Get the concept's rubric, generating + caching it on first use (§2.1)."""
    existing = session.exec(select(Rubric).where(Rubric.concept_id == concept.id)).first()
    if existing is not None:
        return existing
    topic_id = _topic_id_for(session, concept)
    foundation = session.exec(
        select(TopicFoundation).where(TopicFoundation.topic_id == topic_id)).first()
    sources = session.exec(
        select(SourceDoc).where(SourceDoc.topic_id == topic_id,
                                SourceDoc.status == "accepted")).all()
    system, user = build_rubric_prompt(concept.name, foundation, sources)
    resp = llm.complete(system=system, messages=[{"role": "user", "content": user}], tools=[])
    draft = parse_rubric_json(resp.text)
    rubric = Rubric(concept_id=concept.id, criteria=draft.criteria,
                    must_include=draft.must_include, strong_signs=draft.strong_signs,

_GRADE_SYSTEM = (
    "You are a SKEPTICAL grader for an advanced learner's explanation. Grade the "
    "RESPONSE strictly AGAINST the provided rubric/anchor — never from fluency, length, "
    "or confidence. Reward only demonstrated understanding of the listed criteria; "
    "penalize vagueness and missing required elements. Respond with ONLY a JSON object: "
    '{"score": number 0..1, "verdict": str (one sentence), "gaps": [str, specific '

    "self_explanation": "Explain *why* {concept} works, in your own words.",
    "calibration": "Explain *why* {concept} works (we will compare to your confidence).",
    "artifact_review": "Review your shipped artifact for {concept}: what does it "

    "source_quiz": "{question}",

@dataclass
class AssessmentResult:
    assessment: Assessment
    estimate: KnowledgeEstimate
    events: list[AgentEvent]


def build_grade_prompt(concept_name: str, kind: str, criteria: list[str],
                       reference: str, response: str) -> tuple[str, str]:
    lines = [f"CONCEPT: {concept_name}", f"PROBE KIND: {kind}", "", "RUBRIC / ANCHOR:"]
    for c in criteria:
        lines.append(f"- {c}")
    if reference:
        lines.append(f"REFERENCE ANSWER: {reference}")
    lines += ["", "LEARNER RESPONSE:", response or "(empty)"]
    return _GRADE_SYSTEM, "\n".join(lines)


def _deliverable_criteria(session: Session, item_id: int | None) -> list[str]:
    if item_id is None:
        return []
    item = session.get(Item, item_id)
    if item is None:
        return []
    d = session.exec(select(Deliverable).where(Deliverable.phase_id == item.phase_id)).first()
    return d.acceptance_criteria if d else []


def assess_concept(
    session: Session,
    concept: Concept,
    kind: str,
    response: str,
    *,
    llm: LLMClient,
    confidence_predicted: float | None = None,
    baseline_id: int | None = None,
    item_id: int | None = None,
) -> AssessmentResult:
    """Run an anchored probe, persist an Assessment, and update the KnowledgeEstimate."""
    events: list[AgentEvent] = []
    topic_id = _topic_id_for(session, concept)

    def emit(k: str, detail: str = "") -> None:
        ev = AgentEvent(topic_id=topic_id, kind=k, detail=detail)
        session.add(ev)
        events.append(ev)

    # Resolve the anchor (rubric vs. acceptance criteria) and the probe text.
    rubric_id: int | None = None
    must_include: list[str] = []
    weak_signs: list[str] = []
    reference = ""
    if kind == "artifact_review":
        criteria = _deliverable_criteria(session, item_id)
        probe = _PROBE_TEXT[kind].format(concept=concept.name)
    else:
        rubric = ensure_rubric(session, concept, llm=llm)
        rubric_id = rubric.id
        criteria, must_include, weak_signs = rubric.criteria, rubric.must_include, rubric.weak_signs
        probe = _PROBE_TEXT.get(kind, _PROBE_TEXT["self_explanation"]).format(
            concept=concept.name, question=concept.name)
    emit("assess:probe", kind)

    system, user = build_grade_prompt(concept.name, kind, criteria, reference, response)
    grade = parse_grade_json(
        llm.complete(system=system, messages=[{"role": "user", "content": user}], tools=[]).text)

    rules_gaps = apply_rules_checks(response, must_include, weak_signs)
    if rules_gaps:
        emit("assess:rules", f"{len(rules_gaps)} hard gap(s)")
    final_score = penalize_score(grade.score, rules_gaps)
    gaps = list(grade.gaps) + rules_gaps
    emit("assess:graded", f"score={final_score}")

    baseline = session.get(Assessment, baseline_id) if baseline_id else None
    delta = compute_delta(final_score, baseline.score if baseline else None)
    confidence_actual = final_score if kind == "calibration" else grade.confidence_actual

    assessment = Assessment(
        concept_id=concept.id, item_id=item_id, kind=kind, prompt=probe, response=response,
        verdict=grade.verdict, score=final_score, gaps=gaps,
        confidence_predicted=confidence_predicted, confidence_actual=confidence_actual,
        rubric_id=rubric_id, baseline_assessment_id=baseline_id, delta=delta)
    session.add(assessment)

    level, conf = level_from_score(final_score)
    estimate = session.exec(
        select(KnowledgeEstimate).where(KnowledgeEstimate.concept_id == concept.id)).first()
    if estimate is None:
        estimate = KnowledgeEstimate(concept_id=concept.id)
        session.add(estimate)
    estimate.level, estimate.confidence = level, conf
    emit("assess:estimate", f"level={level}")

    emit("assess:done", f"concept {concept.id}")
    session.commit()
    session.refresh(assessment)
    session.refresh(estimate)
    return AssessmentResult(assessment=assessment, estimate=estimate, events=events)
```

import json
from datetime import timedelta

from sqlmodel import Session

from adaptive_academy.llm.types import LLMClient
from adaptive_academy.models import Assessment, Concept, RecallPrompt, SRSCard
from adaptive_academy.models import _now

_RETENTION_SYSTEM = (
    "You propose a FEW high-quality, curated spaced-repetition cards and active-recall "
    "prompts for ONE load-bearing concept, targeting the learner's demonstrated gaps. "
    "Cards test understanding, not trivia. Respond with ONLY a JSON object: "
    '{"cards": [{"front": str, "back": str}], "prompts": [str, free-recall/Feynman '
    'prompts]}. At most 3 cards and 2 prompts. No prose outside the JSON.'
)


def _strip_fences(text: str) -> str:
    cleaned = text.strip()
    if cleaned.startswith("```"):
        cleaned = cleaned.split("```", 2)[1]
        if cleaned.startswith("json"):
            cleaned = cleaned[len("json"):]
    return cleaned.strip()


def propose_retention(
    session: Session, concept: Concept, assessment: Assessment, *, llm: LLMClient,
) -> tuple[list[SRSCard], list[RecallPrompt]]:
    """Generate proposed (user-reviewable) cards + recall prompts from gaps (§2.6)."""
    user = (f"CONCEPT: {concept.name}\nDEMONSTRATED GAPS: "
            f"{'; '.join(assessment.gaps) or '(none)'}\nVERDICT: {assessment.verdict}")
    resp = llm.complete(system=_RETENTION_SYSTEM,
                        messages=[{"role": "user", "content": user}], tools=[])
    data = json.loads(_strip_fences(resp.text))
    cards: list[SRSCard] = []
    for c in (data.get("cards", []) or [])[:3]:
        card = SRSCard(concept_id=concept.id, front=str(c.get("front", "")),
                       back=str(c.get("back", "")), status="proposed", due_at=_now())
        session.add(card)
        cards.append(card)
    prompts: list[RecallPrompt] = []
    for p in (data.get("prompts", []) or [])[:2]:
        rp = RecallPrompt(concept_id=concept.id, prompt=str(p), status="proposed", due_at=_now())
        session.add(rp)
        prompts.append(rp)
    session.commit()
    for c in cards:
        session.refresh(c)
    for p in prompts:
        session.refresh(p)
    return cards, prompts


def grade_card(session: Session, card: SRSCard, result: str) -> SRSCard:
    """Apply SM-2 to a due card and advance its due date."""
    card.ease, card.interval, card.reps = sm2_schedule(
        card.ease, card.interval, card.reps, result)
    card.due_at = _now() + timedelta(days=card.interval)
    card.last_result = result
    session.add(card)
    session.commit()
    session.refresh(card)
    return card
```

```python
# src/adaptive_academy/api/routes_retention.py
from typing import Optional

from fastapi import APIRouter, Depends, HTTPException
from sqlmodel import Session, select

from adaptive_academy.api.deps import get_llm, get_session
from adaptive_academy.api.schemas import (
    CardGradeIn, CardReviewIn, ProposeRetentionIn, RecallPromptOut, RetentionOut, SRSCardOut,
)
from adaptive_academy.engine.retention import grade_card, propose_retention
from adaptive_academy.llm.types import LLMClient
from adaptive_academy.models import (
    Assessment, Concept, RecallPrompt, SRSCard, Track,
)

card_router = APIRouter(prefix="/api/concepts", tags=["retention"])
topic_router = APIRouter(prefix="/api/topics", tags=["retention"])
ops_router = APIRouter(prefix="/api/retention", tags=["retention"])

_VALID_STATUS = {"active", "suspended"}
_VALID_RESULT = {"again", "hard", "good", "easy"}


def _card_out(c: SRSCard) -> SRSCardOut:
    return SRSCardOut(id=c.id, concept_id=c.concept_id, front=c.front, back=c.back,
                      status=c.status, interval=c.interval, reps=c.reps,
                      last_result=c.last_result)


def _prompt_out(p: RecallPrompt) -> RecallPromptOut:
    return RecallPromptOut(id=p.id, concept_id=p.concept_id, prompt=p.prompt, status=p.status)


@card_router.post("/{concept_id}/retention/propose", response_model=RetentionOut)
def propose(concept_id: int, body: ProposeRetentionIn,
            session: Session = Depends(get_session),
            llm: LLMClient = Depends(get_llm)) -> RetentionOut:
    concept = session.get(Concept, concept_id)
    if concept is None:
        raise HTTPException(404, "concept not found")
    if body.from_assessment_id is not None:
        assessment = session.get(Assessment, body.from_assessment_id)
    else:
        assessment = session.exec(
            select(Assessment).where(Assessment.concept_id == concept_id)
            .order_by(Assessment.id.desc())).first()
    if assessment is None:
        raise HTTPException(400, "no assessment to propose retention from")
    cards, prompts = propose_retention(session, concept, assessment, llm=llm)
    return RetentionOut(cards=[_card_out(c) for c in cards],
                        prompts=[_prompt_out(p) for p in prompts])


@topic_router.get("/{topic_id}/retention", response_model=RetentionOut)
def list_retention(topic_id: int, status: Optional[str] = None,
                   session: Session = Depends(get_session)) -> RetentionOut:
    concept_ids = [c.id for c in session.exec(
        select(Concept).join(Track).where(Track.topic_id == topic_id)).all()]
    if not concept_ids:
        return RetentionOut(cards=[], prompts=[])
    cq = select(SRSCard).where(SRSCard.concept_id.in_(concept_ids))
    pq = select(RecallPrompt).where(RecallPrompt.concept_id.in_(concept_ids))
    if status is not None:
        cq = cq.where(SRSCard.status == status)
        pq = pq.where(RecallPrompt.status == status)
    return RetentionOut(
        cards=[_card_out(c) for c in session.exec(cq).all()],
        prompts=[_prompt_out(p) for p in session.exec(pq).all()])


@ops_router.post("/cards/{card_id}/review", response_model=SRSCardOut)
def review_card(card_id: int, body: CardReviewIn,
                session: Session = Depends(get_session)) -> SRSCardOut:
    card = session.get(SRSCard, card_id)
    if card is None:
        raise HTTPException(404, "card not found")
    if body.status not in _VALID_STATUS:
        raise HTTPException(422, "invalid status")
    card.status = body.status
    session.add(card)
    session.commit()
    session.refresh(card)
    return _card_out(card)


@ops_router.post("/cards/{card_id}/grade", response_model=SRSCardOut)
def grade(card_id: int, body: CardGradeIn,
          session: Session = Depends(get_session)) -> SRSCardOut:
    card = session.get(SRSCard, card_id)
    if card is None:
        raise HTTPException(404, "card not found")
    if body.result not in _VALID_RESULT:
        raise HTTPException(422, "invalid result")
    return _card_out(grade_card(session, card, body.result))
```
